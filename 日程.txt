    <script>
    document.addEventListener('DOMContentLoaded', () => {
        lucide.createIcons();

        // ========================================================================
        // SCHEDULE PAGE LOGIC
        // ========================================================================
        const schedulePage = (() => {
            const MOCK_ALL_DAY_TASKS = [
                { title: '喝三杯水', status: 'pending', goalColor: 'indigo' },
                { title: '冥想5分钟', status: 'pending', goalColor: 'purple' },
                { title: '阅读', status: 'pending', goalColor: 'teal' },
            ];

            const MOCK_TIMED_TASKS = [
                { time: '上午 07:30', title: '晨跑3公里', status: 'pending', goalColor: 'green' },
                { time: '上午 09:00', title: '撰写项目周报', status: 'pending', goalColor: 'green' },
                { time: '上午 11:00', title: '回复重要邮件', status: 'pending', goalColor: 'green' },
                { time: '中午 12:00', title: '午餐会议', status: 'completed', goalColor: null },
                { time: '下午 02:30', title: '与客户进行视频会议', description: '讨论第三季度合作方案。', status: 'pending', goalColor: 'purple', isUrgent: true },
                { time: '下午 04:00', title: '团队下午茶', status: 'pending', goalColor: 'green' },
                { time: '下午 05:30', title: '取快递', status: 'completed', goalColor: null },
                { time: '晚上 07:00', title: '健身房锻炼 - 胸部', status: 'pending', goalColor: 'purple' },
            ];

            const colorMap = {
                green: { checkbox: 'border-green-500', bg: 'bg-green-50', text: 'text-green-800' },
                purple: { checkbox: 'border-purple-500', bg: 'bg-purple-50', text: 'text-purple-800' },
                indigo: { checkbox: 'border-indigo-500', bg: 'bg-indigo-50', text: 'text-indigo-800' },
                teal: { checkbox: 'border-teal-500', bg: 'bg-teal-50', text: 'text-teal-800' },
                gray: { checkbox: 'border-gray-400', bg: 'bg-gray-50', text: 'text-gray-500 line-through'}
            };

            function renderAllDayTasks() {
                const container = document.getElementById('all-day-tasks');
                container.innerHTML = `
                    <div class="flex gap-4">
                        <div class="text-xs font-semibold text-blue-600 w-16 text-right pt-2 flex-shrink-0">全天</div>
                        <div class="flex-1 flex flex-wrap gap-2 items-center">
                            ${MOCK_ALL_DAY_TASKS.map(task => {
                                const color = task.status === 'completed' ? 'gray' : (task.goalColor || 'green');
                                return `
                                <div class="task-item inline-flex items-center gap-3 p-1.5 pr-4 rounded-full ${colorMap[color].bg}" data-status="${task.status}" data-goal-color="${task.goalColor || 'green'}">
                                    <button class="task-checkbox w-5 h-5 rounded-full bg-white flex-shrink-0 transition-all flex items-center justify-center ${
                                        task.status === 'completed' ? 'bg-gray-400 text-white' : `border-2 ${colorMap[color].checkbox}`
                                    }">
                                        ${task.status === 'completed' ? '<i data-lucide="check" class="w-3 h-3 -rotate-12"></i>' : ''}
                                    </button>
                                    <p class="task-title text-sm font-medium ${colorMap[color].text}">${task.title}</p>
                                </div>`;
                            }).join('')}
                        </div>
                    </div>`;
            }

            function renderTimedTasks() {
                const container = document.getElementById('schedule-content');
                container.innerHTML = MOCK_TIMED_TASKS.map(task => {
                    const isCompleted = task.status === 'completed';
                    const effectiveColor = isCompleted ? 'gray' : (task.goalColor || 'green');

                    let contentClasses = 'p-3 rounded-lg w-full';
                    if (!isCompleted && task.isUrgent) {
                        contentClasses += ` ${colorMap[effectiveColor].bg} border-2 border-red-600 shadow-sm`;
                    } else {
                        contentClasses += ` ${colorMap[effectiveColor].bg}`;
                    }
                    
                    return `
                    <div class="task-item flex items-start gap-4" data-status="${task.status}" data-goal-color="${task.goalColor || 'green'}" data-is-urgent="${!!task.isUrgent}">
                        <div class="text-sm text-gray-500 w-16 text-right pt-3">${task.time}</div>
                        <div class="flex-1 flex items-start gap-3">
                            <button class="task-checkbox mt-2 w-5 h-5 rounded-full flex-shrink-0 transition-all flex items-center justify-center ${
                                isCompleted ? 'bg-gray-400 text-white' : `border-2 ${task.isUrgent ? 'border-red-600' : colorMap[effectiveColor].checkbox}`
                            }">
                                ${isCompleted ? '<i data-lucide="check" class="w-3 h-3 -rotate-12"></i>' : ''}
                            </button>
                            <div class="task-content ${contentClasses}">
                                <p class="task-title font-semibold ${isCompleted ? colorMap.gray.text : ''}">${task.title}</p>
                                ${task.description ? `<p class="text-sm text-gray-600 mt-1">${task.description}</p>` : ''}
                                ${task.isUrgent ? `
                                <div class="flex items-center gap-2 mt-2 text-xs ${colorMap[effectiveColor].text} font-bold">
                                    <i data-lucide="phone" class="w-3 h-3"></i>
                                    <span>电话预警</span>
                                </div>` : ''}
                            </div>
                        </div>
                    </div>
                `;
                }).join('');
            }

            function initCheckboxes() {
                document.querySelectorAll('#page-home .task-item').forEach(item => {
                    const checkbox = item.querySelector('.task-checkbox');
                    if (!checkbox || checkbox.handler) return;

                    const clickHandler = () => {
                        const isUrgent = item.dataset.isUrgent === 'true';
                        const goalColor = item.dataset.goalColor;
                        const content = item.querySelector('.task-content') || item;
                        const title = item.querySelector('.task-title');

                        if (item.dataset.status === 'pending') {
                            item.dataset.status = 'completed';
                            checkbox.innerHTML = '<i data-lucide="check" class="w-3 h-3 -rotate-12"></i>';
                            lucide.createIcons();
                            checkbox.className = 'task-checkbox mt-2 w-5 h-5 rounded-full flex-shrink-0 transition-all flex items-center justify-center bg-gray-400 text-white';
                            
                            content.className = `task-content ${item.id === 'all-day-tasks' ? '' : 'p-3 rounded-lg w-full'} ${colorMap.gray.bg}`;
                            title.className = `task-title font-semibold ${colorMap.gray.text}`;
                        } else {
                            item.dataset.status = 'pending';
                            checkbox.innerHTML = '';
                            let checkboxColorClass = isUrgent ? 'border-red-600' : colorMap[goalColor].checkbox;
                            checkbox.className = `task-checkbox mt-2 w-5 h-5 rounded-full flex-shrink-0 transition-all flex items-center justify-center border-2 ${checkboxColorClass}`;
                            
                            let contentColorClass = isUrgent ? `${colorMap[goalColor].bg} border-2 border-red-600 shadow-sm` : colorMap[goalColor].bg;
                            content.className = `task-content p-3 rounded-lg w-full ${contentColorClass}`;
                            title.className = `task-title font-semibold`;
                        }
                    };
                    checkbox.handler = clickHandler;
                    checkbox.addEventListener('click', clickHandler);
                });
            }

            function init() {
                renderAllDayTasks();
                renderTimedTasks();
                initCheckboxes();
                lucide.createIcons();

                const header = document.getElementById('schedule-header');
                const allDayTasks = document.getElementById('all-day-tasks');
                function setStickyTop() { if (header && allDayTasks) allDayTasks.style.top = `${header.offsetHeight}px`; }
                setStickyTop();
                window.addEventListener('resize', setStickyTop, { passive: true });

                const weekdayItems = document.querySelectorAll('.weekday-item');
                weekdayItems.forEach(item => {
                    item.addEventListener('click', () => {
                        weekdayItems.forEach(i => {
                            i.classList.remove('bg-blue-600', 'text-white', 'shadow');
                            i.querySelectorAll('span').forEach((s, idx) => s.className = idx === 0 ? 'text-xs text-gray-500' : 'font-semibold mt-1');
                        });
                        item.classList.add('bg-blue-600', 'text-white', 'shadow');
                        item.querySelectorAll('span').forEach((s, idx) => s.className = idx === 0 ? 'text-xs font-light' : 'font-bold text-lg mt-0.5');
                    });
                });
            }
            return { init };
        })();

        // Placeholder for Planning Page Logic
        const planningPage = { init: () => {} };

        // ========================================================================
        // MAIN APP INITIALIZATION
        // ========================================================================
        const navButtons = document.querySelectorAll('.nav-btn');
        const fabButton = document.getElementById('fab-add-task');
        
        navButtons.forEach(button => {
            button.addEventListener('click', () => {
                const pageId = button.dataset.page;
                document.querySelectorAll('main > div[id^="page-"]').forEach(p => p.classList.add('hidden'));
                document.getElementById(`page-${pageId}`).classList.remove('hidden');
                
                navButtons.forEach(btn => {
                    btn.classList.remove('text-blue-600', 'bg-blue-50');
                    btn.classList.add('text-gray-600');
                });
                button.classList.add('text-blue-600', 'bg-blue-50');
                
                fabButton.style.display = (pageId === 'home') ? 'block' : 'none';
            });
        });

        // Initial Load
        document.getElementById('page-planning').classList.add('hidden');
        document.getElementById('page-me').classList.add('hidden');
        document.getElementById('page-home').classList.remove('hidden');
        
        schedulePage.init();
        planningPage.init();
        fabButton.style.display = 'block';
    });
    </script>
