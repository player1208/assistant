    <script>
    document.addEventListener('DOMContentLoaded', () => {

        // =================================================================
        // === 1. 数据管理模块 (Data Management) ============================
        // =================================================================
        const STORAGE_KEY = 'personalAssistantData';
        let appData = { projects: [], activeProjectId: null };

        function saveData() { localStorage.setItem(STORAGE_KEY, JSON.stringify(appData)); }
        function loadData() {
            const savedData = localStorage.getItem(STORAGE_KEY);
            if (savedData) { appData = JSON.parse(savedData); }
            else {
                const defaultProjectId = `proj-${Date.now()}`;
                appData.projects.push({
                    id: defaultProjectId, title: "我的第一个项目",
                    nodes: [ { id: `node-${Date.now()}-1`, title: "示例节点A", x: '10%', y: '20%' }, { id: `node-${Date.now()}-2`, title: "示例节点B", x: '45%', y: '40%' } ],
                    connections: [{ from: `node-${Date.now()}-1`, to: `node-${Date.now()}-2` }]
                });
                appData.activeProjectId = defaultProjectId;
                saveData();
            }
            if (!appData.activeProjectId && appData.projects.length > 0) {
                appData.activeProjectId = appData.projects[0].id;
            }
        }
        function switchProject(projectId) { if (appData.activeProjectId === projectId) return false; appData.activeProjectId = projectId; saveData(); return true; }
        function addNewProject(title) { const newProjectId = `proj-${Date.now()}`; appData.projects.push({ id: newProjectId, title: title, nodes: [], connections: [] }); return newProjectId; }
        function deleteActiveProject() { appData.projects = appData.projects.filter(p => p.id !== appData.activeProjectId); appData.activeProjectId = appData.projects.length > 0 ? appData.projects[0].id : null; saveData(); }
        function updateActiveProjectTitle(newTitle) { const p = appData.projects.find(p => p.id === appData.activeProjectId); if (p) { p.title = newTitle; saveData(); } }
        function updateNodePosition(nodeId, x, y) { const p = appData.projects.find(p => p.id === appData.activeProjectId); if (p) { const n = p.nodes.find(n => n.id === nodeId); if (n) { n.x = x; n.y = y; saveData(); } } }
        function updateNodeTitle(nodeId, newTitle) { const p = appData.projects.find(p => p.id === appData.activeProjectId); if (p) { const n = p.nodes.find(n => n.id === nodeId); if (n) { n.title = newTitle; saveData(); } } }

        // =================================================================
        // === 2. 规划模块 (Planning Canvas Logic) =========================
        // =================================================================
        const planning = (() => {
            const canvas = document.getElementById('plan-canvas');
            const container = document.getElementById('plan-container');
            let state = { lines: [], draggables: [], scale: 1, pan: { x: 0, y: 0 }, isPanning: false, startPoint: { x: 0, y: 0 } };
            
            const updateView = () => { container.style.transform = `translate(${state.pan.x}px, ${state.pan.y}px) scale(${state.scale})`; state.lines.forEach(line => line.position()); };
            
            const init = () => {
                const activeProject = appData.projects.find(p => p.id === appData.activeProjectId);
                if (!activeProject) return;

                const nodes = Array.from(container.querySelectorAll('.plan-node'));
                nodes.forEach(node => {
                    const draggable = new PlainDraggable(node, {
                        containment: container,
                        onDragStart: () => node.classList.add('is-dragging'),
                        onMove: () => state.lines.forEach(line => line.position()),
                        onDragEnd: (pos) => {
                            node.classList.remove('is-dragging');
                            
                            // ================== BUG FIX STARTS HERE ==================
                            // 问题的核心：拖拽库返回的 `pos.left` 和 `pos.top` 是在当前视图（已缩放）下的像素位置。
                            // 我们需要将这个位置换算回未缩放时的“逻辑”位置，然后再存为百分比。
                            
                            // 1. 计算出在未缩放的画布上，节点的真实像素位置。
                            const unscaledLeft = pos.left / state.scale;
                            const unscaledTop = pos.top / state.scale;

                            // 2. 将这个真实像素位置转换为相对于容器宽高的百分比。
                            const newX = `${(unscaledLeft / container.offsetWidth) * 100}%`;
                            const newY = `${(unscaledTop / container.offsetHeight) * 100}%`;
                            // =================== BUG FIX ENDS HERE ===================

                            node.style.left = newX;
                            node.style.top = newY;
                            updateNodePosition(node.id, newX, newY);
                        }
                    });
                    state.draggables.push(draggable);
                });

                if (activeProject.connections) {
                    activeProject.connections.forEach(conn => {
                         const fromEl = document.getElementById(conn.from); const toEl = document.getElementById(conn.to);
                         if (fromEl && toEl) { state.lines.push(new LeaderLine(fromEl, toEl, { color: 'rgba(59, 130, 246, 0.6)', size: 4, path: 'fluid', endPlug: 'arrow1' })); }
                    });
                }
                updateView();
            };

            const cleanup = () => {
                state.lines.forEach(l => l.remove()); state.draggables.forEach(d => d.remove());
                if (container) container.style.transform = '';
                state = { lines: [], draggables: [], scale: 1, pan: { x: 0, y: 0 }, isPanning: false, startPoint: { x: 0, y: 0 } };
            };
            
            const handleMouseDown = (e) => { if (e.target !== canvas && e.target !== container) return; state.isPanning = true; state.startPoint = { x: e.clientX - state.pan.x, y: e.clientY - state.pan.y }; canvas.style.cursor = 'grabbing'; };
            const handleMouseMove = (e) => { if (!state.isPanning) return; e.preventDefault(); state.pan.x = e.clientX - state.startPoint.x; state.pan.y = e.clientY - state.startPoint.y; updateView(); };
            const stopPanning = () => { state.isPanning = false; canvas.style.cursor = 'grab'; };
            const handleWheel = (e) => { e.preventDefault(); const scaleAmount = e.deltaY > 0 ? -0.1 : 0.1; state.scale = Math.max(0.2, Math.min(2, state.scale + scaleAmount)); updateView(); };
            const zoomIn = () => { state.scale = Math.min(2, state.scale + 0.2); updateView(); };
            const zoomOut = () => { state.scale = Math.max(0.2, state.scale - 0.2); updateView(); };
            const zoomReset = () => { state.scale = 1; state.pan = { x: 0, y: 0 }; updateView(); };

            return { init, cleanup, handleMouseDown, handleMouseMove, stopPanning, handleWheel, zoomIn, zoomOut, zoomReset };
        })();


        // =================================================================
        // === 3. UI 渲染模块 (UI Rendering) ===============================
        // =================================================================
        const pages = { home: document.getElementById('page-home'), planning: document.getElementById('page-planning'), me: document.getElementById('page-me') };
        const fabButton = document.getElementById('fab-add-task');

        function renderAll() { renderProjectList(); renderActiveProjectCanvas(); }
        function renderProjectList() {
            const projectListEl = document.getElementById('project-list');
            projectListEl.innerHTML = '';
            appData.projects.forEach(project => {
                const item = document.createElement('button');
                item.className = `project-item w-full text-left p-2 rounded-md text-sm hover:bg-gray-100 ${project.id === appData.activeProjectId ? 'active' : ''}`;
                item.textContent = project.title;
                item.dataset.projectId = project.id;
                projectListEl.appendChild(item);
            });
        }
        function renderActiveProjectCanvas() {
            const activeProject = appData.projects.find(p => p.id === appData.activeProjectId);
            const container = document.getElementById('plan-container');
            const titleInput = document.getElementById('project-title-input');
            const emptyState = document.getElementById('planning-empty-state');
            const canvasWrapper = document.getElementById('plan-canvas-wrapper');
            const planningHeader = document.getElementById('planning-header');
            planning.cleanup(); 
            if (!activeProject) {
                container.innerHTML = ''; titleInput.value = '';
                emptyState.classList.remove('hidden'); canvasWrapper.classList.add('hidden'); planningHeader.classList.add('hidden');
                return;
            }
            emptyState.classList.add('hidden'); canvasWrapper.classList.remove('hidden'); planningHeader.classList.remove('hidden');
            titleInput.value = activeProject.title;
            container.innerHTML = '';
            activeProject.nodes.forEach(node => {
                const nodeEl = document.createElement('div');
                nodeEl.id = node.id;
                nodeEl.className = 'plan-node w-64 bg-white rounded-lg shadow-md p-4';
                nodeEl.style.left = node.x; nodeEl.style.top = node.y;
                nodeEl.innerHTML = `<input type="text" value="${node.title}" data-node-id="${node.id}" class="node-title-input text-lg font-bold w-full bg-transparent focus:outline-none focus:bg-gray-100 rounded-md p-1 -m-1"><p class="text-xs text-gray-500 mt-1">Details about the node...</p><button class="open-subtasks-btn mt-3 text-sm text-blue-600 hover:underline">查看子任务</button>`;
                container.appendChild(nodeEl);
            });
            planning.init();
        }
        function switchPage(pageId) {
            const currentPageId = document.querySelector('.nav-btn.text-blue-600').dataset.page;
            if (currentPageId === 'planning' && pageId !== 'planning') { planning.cleanup(); }
            Object.values(pages).forEach(page => page.classList.add('hidden'));
            pages[pageId].classList.remove('hidden');
            document.querySelectorAll('.nav-btn').forEach(btn => { btn.classList.remove('text-blue-600', 'bg-blue-50'); btn.classList.add('text-gray-600'); });
            document.querySelector(`.nav-btn[data-page="${pageId}"]`).classList.add('text-blue-600', 'bg-blue-50');
            fabButton.style.display = (pageId === 'home') ? 'block' : 'none';
            if (pageId === 'planning') { renderAll(); }
        }

        // =================================================================
        // === 4. 应用主入口和事件监听 (App Initializer & Listeners) =======
        // =================================================================
        function initialize() {
            loadData();
            switchPage('home'); 
            setupEventListeners();
            lucide.createIcons();
        }

        function setupEventListeners() {
            document.querySelectorAll('.nav-btn').forEach(button => { button.addEventListener('click', () => { switchPage(button.dataset.page); }); });
            document.getElementById('project-list').addEventListener('click', (e) => { const item = e.target.closest('.project-item'); if (item) { if (switchProject(item.dataset.projectId)) { renderAll(); } } });
            
            const addProjectModal = document.getElementById('add-project-modal');
            document.getElementById('add-project-btn').onclick = () => addProjectModal.classList.remove('hidden');
            document.getElementById('cancel-add-project').onclick = () => addProjectModal.classList.add('hidden');
            document.getElementById('confirm-add-project').onclick = () => { const input = document.getElementById('new-project-name'); const newName = input.value.trim(); if (newName) { const newId = addNewProject(newName); switchProject(newId); renderAll(); input.value = ''; addProjectModal.classList.add('hidden'); } };
            
            const deleteProjectModal = document.getElementById('delete-project-modal');
            document.getElementById('delete-project-btn').onclick = () => { if (appData.activeProjectId) deleteProjectModal.classList.remove('hidden'); };
            document.getElementById('cancel-delete-project').onclick = () => deleteProjectModal.classList.add('hidden');
            document.getElementById('confirm-delete-project').onclick = () => { deleteActiveProject(); renderAll(); deleteProjectModal.classList.add('hidden'); };
            
            document.getElementById('project-title-input').addEventListener('change', (e) => { updateActiveProjectTitle(e.target.value); renderProjectList(); });
            document.getElementById('plan-container').addEventListener('change', (e) => { if (e.target.matches('.node-title-input')) { updateNodeTitle(e.target.dataset.nodeId, e.target.value); } });
            
            const canvas = document.getElementById('plan-canvas');
            canvas.addEventListener('mousedown', planning.handleMouseDown);
            canvas.addEventListener('mousemove', planning.handleMouseMove);
            canvas.addEventListener('mouseup', planning.stopPanning);
            canvas.addEventListener('mouseleave', planning.stopPanning);
            canvas.addEventListener('wheel', planning.handleWheel);
            document.getElementById('zoom-in-btn').onclick = planning.zoomIn;
            document.getElementById('zoom-out-btn').onclick = planning.zoomOut;
            document.getElementById('zoom-reset-btn').onclick = planning.zoomReset;
        }

        initialize();
    });
